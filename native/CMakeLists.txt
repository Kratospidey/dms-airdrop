cmake_minimum_required(VERSION 3.20)
project(dms_airdrop_picker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6KIO 6.0 REQUIRED)
find_package(KF6IconThemes 6.0 REQUIRED)
add_executable(dms-airdrop-picker picker.cpp)
target_link_libraries(dms-airdrop-picker PRIVATE Qt6::Widgets KF6::KIOFileWidgets KF6::IconThemes)
target_compile_options(dms-airdrop-picker PRIVATE -Wall -Wextra -Wpedantic)
option(DMS_PICKER_TESTS "Build isolated native selection harness" OFF)
if(DMS_PICKER_TESTS)
    add_executable(picker-selection-test picker.cpp ../tests/native_selection.cpp)
    target_compile_definitions(picker-selection-test PRIVATE DMS_PICKER_LIBRARY)
    target_link_libraries(picker-selection-test PRIVATE Qt6::Widgets KF6::KIOFileWidgets KF6::IconThemes)
endif()
